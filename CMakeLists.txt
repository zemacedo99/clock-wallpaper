cmake_minimum_required(VERSION 3.16)
project(wallclock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(wallclock_core src/core.cpp src/config.cpp src/renderer.cpp src/codecs.cpp src/bridge.cpp src/scheduler.cpp)
set_target_properties(wallclock_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(wallclock_core PUBLIC include PRIVATE third_party)
option(WALLCLOCK_BUILD_DESKTOP "Build desktop CLI and tests (disable for mobile hosts)" ON)
if(NOT WALLCLOCK_BUILD_DESKTOP)
  return()
endif()
if(WIN32)
  set(ADAPTER src/platform/windows.cpp)
elseif(APPLE AND NOT IOS)
  enable_language(OBJCXX)
  set(ADAPTER src/platform/macos.mm)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID)
  set(ADAPTER src/platform/linux.cpp)
else()
  set(ADAPTER src/platform/unsupported.cpp)
endif()
add_executable(wallclock src/main.cpp ${ADAPTER})
target_link_libraries(wallclock PRIVATE wallclock_core)
if(WIN32)
  target_link_libraries(wallclock PRIVATE user32 advapi32)
  add_executable(wallclock_startup WIN32 src/windows_supervisor.cpp)
  target_include_directories(wallclock_startup PRIVATE include)
  target_link_libraries(wallclock_startup PRIVATE user32 advapi32 shell32)
elseif(APPLE AND NOT IOS)
  target_link_libraries(wallclock PRIVATE "-framework AppKit")
endif()
enable_testing()
add_executable(wallclock_tests tests/tests.cpp)
target_link_libraries(wallclock_tests PRIVATE wallclock_core)
add_test(NAME core_renderer COMMAND wallclock_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples/minimal.ini)
if(WIN32)
  add_executable(wallclock_events_tests tests/windows_events.cpp src/platform/windows.cpp)
  target_include_directories(wallclock_events_tests PRIVATE include)
  target_link_libraries(wallclock_events_tests PRIVATE user32 advapi32)
  add_test(NAME native_events COMMAND wallclock_events_tests)
  add_executable(wallclock_supervision_tests tests/windows_supervision.cpp)
  target_include_directories(wallclock_supervision_tests PRIVATE include)
  target_link_libraries(wallclock_supervision_tests PRIVATE shell32)
  add_test(NAME supervision COMMAND wallclock_supervision_tests)
endif()
